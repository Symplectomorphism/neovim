\begin{multicols}{2}

\section{General \& Navigation}
\begin{description}
  \item[\key{Esc}] Clear search highlight
  \item[\key{Esc Esc}] Leave terminal mode (Claude, Antigravity, any \texttt{:terminal})
  \item[\key{C-h/j/k/l}] Move focus between window splits
  \item[\key{C-o} / \key{C-i}] Jump back / forward (after \key{grd}, searches, etc.)
  \item[\key{leader q}] Diagnostics to quickfix list
  \item[\key{zR} / \key{zM}] Open / close all folds
  \item[\key{za} / \key{zo} / \key{zc}] Toggle / open / close fold under cursor
  \item[\key{saiw)}] Surround Add Inner Word with \texttt{)}
  \item[\key{sd'}] Surround Delete \texttt{'}
  \item[\key{sr)'}] Surround Replace \texttt{)} with \texttt{'}
  \item[\key{va)} / \key{ci'}] Select / change around-or-inside a text object
  \item[\key{\textbackslash}] Reveal current file in neo-tree
\end{description}

\section{Built-in Essentials}
\begin{description}
  \item[\key{u} / \key{C-r}] Undo / redo
  \item[\key{.}] Repeat last change
  \item[\key{C-v}] Visual block mode (column edits)
  \item[\key{qa ... q}] Record macro to register \texttt{a}
  \item[\key{\@a}] Play macro from register \texttt{a}
  \item[\key{ma} / \key{`a}] Set mark \texttt{a} / jump to mark \texttt{a}
  \item[\key{J}] Join line below with a space
  \item[\key{C-w s} / \key{C-w v}] Split window horizontally / vertically
  \item[\key{C-w q}] Close current window
\end{description}

\section{Search (Telescope) --- \key{leader s...}}
\begin{description}
  \item[\key{sf}] Find files
  \item[\key{sg}] Live grep
  \item[\key{sw}] Grep word under cursor
  \item[\key{sd}] Search diagnostics
  \item[\key{sh}] Search help
  \item[\key{sk}] Search keymaps
  \item[\key{ss}] Search Telescope pickers
  \item[\key{sr}] Resume last search
  \item[\key{s.}] Recent files
  \item[\key{s/}] Grep in open files only
  \item[\key{sn}] Find files in nvim config
  \item[\key{leader leader}] Find open buffers
  \item[\key{leader /}] Fuzzy search current buffer
\end{description}

\section{Debugging (nvim-dap) --- Python \& C++}
\begin{description}
  \item[\key{F5}] Continue / start
  \item[\key{F1} / \key{F2} / \key{F3}] Step into / over / out
  \item[\key{leader b}] Toggle breakpoint
  \item[\key{leader B}] Conditional breakpoint
  \item[\key{F7}] Toggle debugger UI
\end{description}
{\small\color{note} First use in a session loads the plugin and may pause briefly while Mason finishes registering.}

\section{Julia REPL (iron.nvim) --- \texttt{.jl} files}
\begin{description}
  \item[\key{localleader is}] Open REPL
  \item[\key{localleader ir}] Restart REPL
  \item[\key{localleader if}] Focus REPL (from code) / jump back (from REPL)
  \item[\key{localleader ih}] Hide REPL
  \item[\key{localleader sl}] Send line
  \item[\key{localleader sp}] Send paragraph
  \item[\key{localleader sc}] Send selection / motion
  \item[\key{localleader sf}] Send whole file
  \item[\key{localleader su}] Send until cursor
  \item[\key{localleader sq}] Exit REPL
  \item[\key{localleader scl}] Clear REPL
\end{description}
{\small\color{note} Real terminal REPL: Pkg mode \texttt{]}, help \texttt{?}, shell \texttt{;}, and OhMyREPL all work natively.}

\section{Python REPL (iron.nvim) --- \texttt{.py} files}
\begin{description}
  \item[\key{localleader is}] Open REPL (ipython)
  \item[\key{localleader ir}] Restart REPL
  \item[\key{localleader if}] Focus REPL (from code) / jump back (from REPL)
  \item[\key{localleader ih}] Hide REPL
  \item[\key{localleader sl}] Send line
  \item[\key{localleader sp}] Send paragraph
  \item[\key{localleader sc}] Send selection / motion
  \item[\key{localleader sb}] Send cell/block (between \texttt{\# \%\%} markers)
  \item[\key{localleader sf}] Send whole file
  \item[\key{localleader su}] Send until cursor
  \item[\key{localleader sq}] Exit REPL
  \item[\key{localleader scl}] Clear REPL
  \item[\key{localleader rr}] Run file non-interactively (floating term)
  \item[\key{]]} / \key{[[}] Next / previous cell (\texttt{\# \%\%} marker)
\end{description}
{\small\color{note} \texttt{\# \%\%} cells are the Jupytext/VS Code "percent format" --- same convention \texttt{.ipynb} files already round-trip through. \key{localleader rr} is a one-shot run, separate from the REPL. Both auto-detect a project \texttt{.venv} (e.g. \texttt{uv venv}), walking up from the file, and fall back to the global interpreter if none is found.}

\section{Git Hunks (gitsigns)}
\begin{description}
  \item[\key{]c} / \key{[c}] Next / previous hunk
  \item[\key{leader hs}] Stage hunk (visual: stage selection)
  \item[\key{leader hr}] Reset hunk (visual: reset selection)
  \item[\key{leader hS}] Stage whole buffer
  \item[\key{leader hR}] Reset whole buffer
  \item[\key{leader hu}] Undo last stage
  \item[\key{leader hp}] Preview hunk diff
  \item[\key{leader hb}] Blame current line
  \item[\key{leader hd}] Diff file against index
  \item[\key{leader hD}] Diff file against last commit
  \item[\key{leader tb}] Toggle inline blame
  \item[\key{leader tD}] Toggle inline preview of deleted lines
\end{description}

\section{AI Agents --- \key{leader a}}
\begin{description}
  \item[\key{ac}] Toggle Claude Code
  \item[\key{af}] Focus Claude (smart toggle)
  \item[\key{ar}] Resume Claude session
  \item[\key{aC}] Continue Claude session
  \item[\key{am}] Select Claude model
  \item[\key{ab}] Add current buffer to context
  \item[\key{as}] Send selection (visual) / add file (tree)
  \item[\key{aa} / \key{ad}] Accept / deny Claude's diff
  \item[\key{ag}] Toggle Antigravity CLI (\texttt{agy})
\end{description}
{\small\color{note} After a Claude/Antigravity window gains focus you're dropped into terminal mode --- \key{Esc Esc} to leave it, \key{i} to resume typing to it.}

\section{Quarto / Molten --- \texttt{.qmd} notebooks}
\begin{description}
  \item[\key{localleader rc}] Run cell
  \item[\key{localleader rl}] Run line
  \item[\key{localleader rA}] Run all cells
  \item[\key{localleader r}] (visual) Run range
  \item[\key{localleader mi} / \key{md}] Init / stop Molten kernel
  \item[\key{localleader mp}] Image popup
  \item[\key{localleader mb}] Open output in browser
  \item[\key{localleader mh} / \key{ms}] Hide / show output
\end{description}
{\small\color{note} \texttt{.ipynb} opens via jupytext as \texttt{ft=python} (\texttt{\# \%\%} cells), \emph{not} quarto --- rc/rl/rA don't bind there. Use \texttt{:MoltenEvaluateLine}/\texttt{Operator}/\texttt{Visual} for kernel output with plots, or the Python REPL section's \key{localleader sb}/\key{sc}/etc. (iron.nvim) for plain-text ipython output.}
